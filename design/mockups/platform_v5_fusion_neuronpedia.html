def sweep_ar(model, K=16, arms=("is_a", "attr", "syntax", "rand")):
    share = exact_additive_budget(model)          # 铁律 (a) 精确可加向量预算
    for arm in arms:
        E = [share.write(arm, k=k) for k in range(1, K+1)]
        rel = (E[0] - E) / E[0]                  # S_rel 归一化
        yield arm, rel
